import { Card, CardGrid, LinkButton } from "@astrojs/starlight/components"

```ts import.meta.vitest
import {
  Actor,
  Document,
  Documents,
  DocumentStorage
} from "@jpowersdev/effect-documents"
import { Effect, Layer, Option } from "effect"

const program = Effect.gen(function*() {
  const documents = yield* Documents.Documents
  const actor: Actor.Actor = {
    by: Actor.Id.make("@owner"),
    via: Option.none()
  }
  const id = Document.Id.make("performance-map")

  const first = yield* documents.create({
    id,
    actor,
    title: "Performance map",
    body: "Clicks take 40 ms."
  })
  yield* documents.apply({
    id,
    actor,
    command: {
      _tag: "Edit",
      oldString: "40 ms",
      newString: "32 ms"
    }
  })

  // The current text, and any earlier version
  const now = yield* documents.read(id)
  const then = yield* documents.read(id, {
    version: first.version
  })
  yield* Effect.log([now.body, then.body])
})

const DocumentsLive = Documents.layer.pipe(Layer.provide(DocumentStorage.layerMemory))

await Effect.runPromise(program.pipe(Effect.provide(DocumentsLive)))
```

<CardGrid>
  <Card title="Attributed history" icon="document">
    Every successful command appends one immutable revision, crediting who ran it. Any version can be read back.

    ```ts
    yield* documents.apply(request)
    yield* documents.history(id)
    yield* documents.read(id)
    ```

    <LinkButton href="guides/commands/" variant="minimal" icon="right-arrow">Commands</LinkButton>
  </Card>
  <Card title="Anchored review" icon="comment">
    Discussions attach to the text they're about, and follow it through later edits.

    ```ts
    const edited = yield* documents.apply(request)
    const thread = edited.threads[0]!
    thread.range // Follows its text through the edit
    ```

    <LinkButton href="guides/commands/#discuss-a-passage" variant="minimal" icon="right-arrow">Discussions</LinkButton>
  </Card>

    ```ts
    const text = yield* ReplicatedText.makeConfirmed(options)

    yield* documents.apply(request)
    ```
